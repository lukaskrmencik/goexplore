--------------------------------------------------------------------------------

    <testsuites>
        <testsuite name="Unit">
            <directory>tests/Unit</directory>
        </testsuite>
        <testsuite name="Feature">
            <directory>tests/Feature</directory>
        </testsuite>
    </testsuites>
    <php>
        <env name="APP_ENV" value="testing"/>
        <env name="DB_CONNECTION" value="sqlite"/>
        <env name="DB_DATABASE" value=":memory:"/> <!-- Testy běží v RAM -->
        <env name="QUEUE_CONNECTION" value="sync"/> <!-- Joby se vykonají ihned -->
    </php>
</phpunit>

6. API ROZHRANÍ: Definice Rout (Backend PHP)
Soubor: backend-php/routes/api.php
Popis: Kompletní seznam endpointů dostupných pro frontend. Používá `JwtMiddleware` pro zabezpečené sekce.
--------------------------------------------------------------------------------
// Veřejné
Route::post('/signup', [AuthController::class, 'signup']);
Route::post('/login', [AuthController::class, 'login']);

// Zabezpečené (vyžadují Bearer Token)
Route::middleware('jwt')->group(function () {
    
    // Uživatel
    Route::get('/users/my', [UserController::class, 'singleUser']);
    Route::patch('/users/my/location', [UserController::class, 'setUserLocation']);

    // Trasy (CRUD + Akce)
    Route::post('/routes', [RouteController::class, 'createRoute']);
    Route::post('/routes/list', [RouteController::class, 'allRoutes']);
    Route::post('/routes/{id}/calculate', [RouteController::class, 'calculateRoute']); // Spuštění jobu
    Route::get('/routes/job/{jobId}/progress', ...); // Polling stavu

    // Vybavení trasy
    Route::post('/routes/{id}/equipment', [RouteEquipmentController::class, 'addEquipment']);
    
    // Body zájmu a Kempy (Detail)
    Route::get('/camps/{id}', [CampController::class, 'singleCamp']);
    Route::get('/poi/{id}', [PoiController::class, 'singlePoi']);
});
