{
    protected $fillable = [
        'users_id', 'mode', 'name', 'start', 'end', 'start_date', 'end_date',
        'axis', 'complete_route', 'buffer_size', 'max_route_length_day', 'poi_per_day'
    ];

    // Automatické přetypování geometrických dat na objekty Point/LineString
    protected $casts = [
        'start' => Point::class,
        'end' => Point::class,
        'axis' => LineString::class,
        'complete_route' => LineString::class,
    ];

    // Relace na kempy (N:M přes routes_camps)
    public function camps()
    {
        return $this->belongsToMany(Camp::class, 'routes_camps', 'routes_id', 'camps_id')
            ->withPivot('order')
            ->orderBy('pivot_order')
            ->withTimestamps();
    }

    // Relace na body zájmu (N:M přes routes_poi)
    public function poi()
    {
        return $this->belongsToMany(Poi::class, 'routes_poi', 'routes_id', 'poi_id')
            ->withPivot(['order', 'routes_clusters_id'])
            ->orderBy('pivot_order')
            ->withTimestamps();
    }
    
    // ... další relace (users, waypoints, equipment) ...
}

def choose_camps(ideal_camps, camps, config_scores, days, DEFAULT_CAMP_SEASON):
    # Rozdělení kempů do skupin podle nejbližšího ideálního bodu
    camp_groups = camp_groups_by_ideal_camp(camps, ideal_camps, days, DEFAULT_CAMP_SEASON)

    # Skórování a výběr nejlepšího kempu v každé skupině
    best_camps = []
    for camp_group in camp_groups:
        camp_scores = []
        for camp in camp_group["camps"]:
            # Normalizace hodnot a výpočet skóre
            score = calc_camp_score(camp, config_scores, ...)
            camp_scores.append({"camp": camp, "score": score})
        
        # Výběr kempu s max skóre
        best_camp = max(camp_scores, key=lambda x: x["score"])
        best_camps.append(best_camp)

    return best_camps

        }, 2000); // Dotaz každé 2 sekundy
    };

    return { route, nextStep, prevStep, startCalculation, calculationProgress, ... };
