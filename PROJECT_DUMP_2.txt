3. EDITOR MAPY: Hlavní Hook (Frontend React)
Soubor: frontend/src/features/mapEditor/hooks/useMapEditor.ts
Popis: Hook `useMapEditor` načítá data trasy (včetně geometrie `complete_route`) a inicializuje pod-hooky pro správu POI a clusterů (`usePois`). Slouží jako datová vrstva pro mapový editor.
--------------------------------------------------------------------------------
export const useMapEditor = (routeId: number | null) => {
    const [route, setRoute] = useState<Route | undefined>(undefined);
    const [routeLine, setrouteLine] = useState<LineString | undefined>(undefined);
    const { pois, clusters } = usePois(route); // Načtení bodů a shluků

    useEffect(() => {
        const fetchMarkers = async () => {
            if(routeId !== null){
                setLoading(true);
                try {
                    const routeData = await fetchGetRoute(routeId);
                    setRoute(routeData);
                    setrouteLine(routeData.complete_route); // Nastavení čáry trasy
                } finally {
                    setLoading(false);
                }
            }
        };
        fetchMarkers();
    }, []);

    return { route, loading, error, pois, camps, clusters, routeLine };
}

4. EDITOR MAPY: Klikání do mapy (Frontend React)
Soubor: frontend/src/features/editors/routeAxisEditor/components/MapClickHandler.tsx
Popis: Komponenta, která se "připojí" k mapě pomocí `useMapEvents` (z React-Leaflet). Odchytává událost `click` a pokud je editor aktivní (`isActive`), předává souřadnice parent komponentě. Používá se pro kreslení osy trasy.
--------------------------------------------------------------------------------
const MapClickHandler: React.FC<MapClickHandlerProps> = ({ onMapClick, isActive }) => {
    useMapEvents({
        click(e) {
            if (isActive) {
                onMapClick(e.latlng.lat, e.latlng.lng);
            }
        },
    });
    return null; // Nevykresluje nic, jen poslouchá eventy
};

6. BEZPEČNOST: JWT Middleware (Backend PHP)
Soubor: backend-php/app/Http/Middleware/JwtMiddleware.php
Popis: Middleware, který pro každý request ověřuje platnost JWT tokenu. Pokud token chybí nebo je neplatný, vrací 401 Unauthorized. Používá knihovnu `tymon/jwt-auth`.
--------------------------------------------------------------------------------
class JwtMiddleware
{
    public function handle(Request $request, Closure $next)
    {
        try {
            JWTAuth::parseToken()->authenticate();
        } catch (Exception $e) {
            return response()->error('Unauthorized', 401);
        }
        return $next($request);
    }
}

10. GEO VÝPOČTY: Frontend Utils (TypeScript)
Soubor: frontend/src/utils/geo.ts
Popis: Pomocné funkce pro konverzi mezi GeoJSON (používaný v datech) a Leaflet LatLng (používaný v mapě).
--------------------------------------------------------------------------------
export const geojsonPointToLatLng = (point: Point): LatLngExpression => {
    const coords = point.coordinates;
    return [coords[1], coords[0]]; // Leaflet chce [lat, lng], GeoJSON má [lng, lat]
};

export const geoJsonLineStringToLatLng = (lineString: LineString): LatLngExpression[] => {
    const coords = lineString.coordinates;
    return coords.map(coord => [coord[1], coord[0]]);
};

11. GEO VÝPOČTY: Backend Service (PHP)
Soubor: backend-php/app/Services/GeoService.php
Popis: Jednoduchá validace geografických dat na backendu. Kontroluje, zda osa trasy (axis) začíná a končí ve stejných bodech jako definovaný start a cíl trasy.
--------------------------------------------------------------------------------
class GeoService
{
    public static function startEndMatchAxis(array $startPoint, array $endPoint, array $axis): ?string
    {
        $firstAxisPoint = $axis[0];
        $lastAxisPoint = $axis[count($axis) - 1];

        if ($firstAxisPoint !== $startPoint) return 'First coordinate...';
        if ($lastAxisPoint !== $endPoint) return 'Last coordinate...';
        return null;
    }
}
